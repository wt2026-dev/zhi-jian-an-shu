### 1.3 乐观锁更新

```javascript
async function updateWorkOrder(id, updates, expectedVersion) {
  const keys = Object.keys(updates);
  const sets = keys.map((k, i) => `${k}=$${i + 1}`).join(', ');
  const result = await db.query(`
    UPDATE work_orders SET ${sets},
        version = version + 1, updated_at = NOW()
    WHERE id = $${keys.length + 1} AND version = $${keys.length + 2}
  `, [...Object.values(updates), id, expectedVersion]);
  if (result.rowCount === 0) throw new VersionConflictError(id);
  return result.rows[0];
}
```

function extractAiJson(raw: string): unknown {
  let s = raw.trim();
  if (s.startsWith('```json')) s = s.slice(7);
  if (s.startsWith('```'))     s = s.slice(3);
  if (s.endsWith('```'))       s = s.slice(0, -3);
  try { return JSON.parse(s); } catch { throw new AiResponseParseError(raw); }
}

### 3.3 AI 错误处理

```typescript
function handleAiRouteError(error, res) {
  if (error instanceof AiResponseParseError) {
    return res.status(502).json({ code: 'AI_PARSE_ERROR', message: 'AI响应格式异常', rawResponse: error.rawResponse });
  }
  if (error.status === 429) {
    return res.status(429).json({ code: 'RATE_LIMITED', message: 'AI服务繁忙，请稍后重试', retryAfter: error.retryAfter });
  }
  return res.status(500).json({ code: 'AI_SERVICE_ERROR', message: 'AI服务暂时不可用' });
}
```

### 3.4 流式接口实现（SSE）

```typescript
// src/server/routes/ai.ts
router.post('/chat/stream', auth, async (req, res) => {
  res.writeHead(200, {
    'Content-Type': 'text/event-stream',
    'Cache-Control': 'no-cache',
    'Connection': 'keep-alive',
  });

  try {
    const stream = await client.chat.completions.create({
      model: 'qwen-plus',
      messages: req.body.messages,
      stream: true,
    });
    for await (const chunk of stream) {
      const delta = chunk.choices?.[0]?.delta?.content ?? '';
      if (delta) res.write(`data: ${JSON.stringify({ type: 'token', content: delta })}\n\n`);
    }
    res.write('data: [DONE]\n\n');
    res.end();
  } catch (error) {
    res.write(`data: ${JSON.stringify({ type: 'error', message: error.message })}\n\n`);
    res.write('data: [DONE]\n\n');
    res.end();
  }
});
```

### 4.1 DatabaseAdapter 双模式

```typescript

let adapter: DatabaseAdapter;

export async function getDb(): Promise<DatabaseAdapter> {
  if (adapter) return adapter;

```

PGlite 路径：`data/.pglite-active.json` 记录活跃目录；多个脚本并发会冲突，**同一时刻只允许一个 app 实例占用 PGlite 目录**（`.pglite-active.json` 用 PID + 启动时间戳仲裁）。

### 4.2 迁移执行（幂等吞 already exists）

```typescript
async function runMigrations(adapter: DatabaseAdapter) {
  const dir = path.join(__dirname, 'migrations');
  const files = (await fs.readdir(dir)).filter(f => f.endsWith('.sql')).sort();

  for (const file of files) {
    const sql = await fs.readFile(path.join(dir, file), 'utf-8');
    const statements = sql.split(/;\s*$/m).filter(s => s.trim());
    for (const stmt of statements) {
      try { await adapter.query(stmt); }
      catch (e) {
        // 吞 "already exists" 类错误，确保重启 / 复用同一目录不会炸
        if (!String(e.message).match(/already exists/i)) throw e;
      }
    }
  }
}
```

**没有 `schema_migrations` 表**（老文档说有的，那是设计目标，实现没做 —— 靠 "CREATE TABLE IF NOT EXISTS" + 吞 already exists 实现幂等）。25 个迁移文件位于 `src/db/migrations/`，编号 001-025。

## 5 前端数据层

### 5.1 自研 queryClient（替代 TanStack Query）

```typescript
// src/utils/queryClient.ts
class QueryClient {
  private cache = new Map<string, { data: unknown; ts: number; staleTime: number }>();
  async fetch(key: string, fetcher: () => Promise<unknown>, staleTime = 30_000) {
    const hit = this.cache.get(key);
    if (hit && Date.now() - hit.ts < hit.staleTime) return hit.data;
    const data = await fetcher();
    this.cache.set(key, { data, ts: Date.now(), staleTime });
    return data;
  }
  invalidate(prefix: string) { /* 删除匹配前缀的缓存 */ }
}
```

`staleTime` 默认 30s，错误重试 2 次，`refetchOnWindowFocus: false`。乐观更新回滚监听 `VERSION_CONFLICT`。

### 5.2 乐观更新实现

```typescript
async function updateOrder(id, updates) {
  const previous = queryClient.get(['order', id]);
  queryClient.set(['order', id], { ...previous, ...updates, version: previous.version + 1 });
  try {
    return await api.put(`/workorders/${id}`, { ...updates, version: previous.version });
  } catch (e) {
    if (e.code === 'VERSION_CONFLICT') {
      queryClient.set(['order', id], previous);
      toast.error('数据已被修改，请刷新后重试');
    }
    throw e;
  } finally {
    queryClient.invalidate(['orders']);
  }
}
```

export const ROLE_PERMISSIONS: Record<Role, string[]> = {
  admin:     ['*'],
  manager:   ['order:read', 'order:write', 'order:close', 'ticket:approve', 'report:read', ...],
  tech1:     ['order:read', 'order:work', 'order:comment', 'ticket:sign', 'ai:diagnose', ...],
  tech2:     ['order:read', 'order:work', 'order:comment', 'ticket:sign'],
  inspector: ['inspection:read', 'inspection:write', 'order:create', ...],
  operator:  ['order:create', 'order:read', 'ticket:create'],
  archivist: ['knowledge:read', 'knowledge:write', 'knowledge:delete', 'rag:query'],
};

export function can(role: Role, perm: string): boolean {
  const list = ROLE_PERMISSIONS[role] ?? [];
  if (list.includes('*')) return true;
  return list.includes(perm);
}
