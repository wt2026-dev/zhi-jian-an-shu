### 1.3 乐观锁更新

```javascript
async function updateWorkOrder(id, updates, expectedVersion) {
  const result = await db.query(`
    UPDATE work_orders 
    SET ${Object.keys(updates).map((k, i) => `${k}=$${i + 1}`).join(', ')},
        version = version + 1,
        updated_at = NOW()
    WHERE id = $${Object.keys(updates).length + 1} 
      AND version = $${Object.keys(updates).length + 2}
  `, [...Object.values(updates), id, expectedVersion]);

  if (result.rowCount === 0) {
    throw new VersionConflictError(id);
  }
  return result.rows[0];
}
```

## 3 AI 服务层实现

### 3.1 DashScope 调用封装

```javascript
const { DashScope } = require('dashscope-sdk');

async function callDashScopeJson(messages, options = {}) {
  const response = await DashScope.chat.completions.create({
    model: options.model || 'qwen-plus',
    messages,
    temperature: options.temperature || 0.7,
    max_tokens: options.maxTokens || 2048,
    response_format: { type: 'json_object' }
  });
  return response.choices[0].message.content;
}
```

### 3.2 AI 响应解析

```javascript
function extractAiJson(rawResponse) {
  try {
    // 处理可能的 markdown 代码块包裹
    let cleaned = rawResponse.trim();
    if (cleaned.startsWith('```json')) {
      cleaned = cleaned.slice(7);
    }
    if (cleaned.endsWith('```')) {
      cleaned = cleaned.slice(0, -3);
    }
    return JSON.parse(cleaned);
  } catch (error) {
    throw new AiResponseParseError(rawResponse);
  }
}
```

### 3.3 错误处理

```javascript
function handleAiRouteError(error, res) {
  if (error instanceof AiResponseParseError) {
    return res.status(502).json({
      code: 'AI_PARSE_ERROR',
      message: 'AI响应格式异常',
      rawResponse: error.rawResponse
    });
  }
  if (error.status === 429) {
    return res.status(429).json({
      code: 'RATE_LIMITED',
      message: 'AI服务繁忙，请稍后重试',
      retryAfter: error.retryAfter
    });
  }
  return res.status(500).json({
    code: 'AI_SERVICE_ERROR',
    message: 'AI服务暂时不可用'
  });
}
```

### 4.2 迁移执行

```javascript
async function runMigrations() {
  const migrations = readdirSync('./migrations')
    .filter(f => f.endsWith('.sql'))
    .sort();
  
  for (const migration of migrations) {
    const applied = await db.query(
      'SELECT 1 FROM schema_migrations WHERE version = $1',
      [migration]
    );
    if (applied.rowCount === 0) {
      const sql = readFileSync(`./migrations/${migration}`, 'utf-8');
      await db.query(sql);
      await db.query(
        'INSERT INTO schema_migrations (version) VALUES ($1)',
        [migration]
      );
    }
  }
}
```

### 5.2 乐观更新实现

```javascript
useMutation({
  mutationFn: updateOrder,
  onMutate: async (newData) => {
    await queryClient.cancelQueries(['order', newData.id]);
    const previous = queryClient.getQueryData(['order', newData.id]);
    queryClient.setQueryData(['order', newData.id], old => ({
      ...old, ...newData, version: old.version + 1
    }));
    return { previous };
  },
  onError: (err, newData, context) => {
    // 409 冲突时回滚
    if (err.code === 'VERSION_CONFLICT') {
      queryClient.setQueryData(['order', newData.id], context.previous);
      toast.error('数据已被修改，请刷新后重试');
    }
  },
  onSettled: () => {
    queryClient.invalidateQueries(['orders']);
  }
});
```
